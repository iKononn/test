class SerialPort {
public:
    virtual ~SerialPort() = default;
    virtual HANDLE openSerialPort(const wstring& portName, DWORD baudRate) {
        return INVALID_HANDLE_VALUE;
    }

    virtual void sendMessage(HANDLE hSerial, const string& message) {
        SetLastError(ERROR_IO_DEVICE);
    }

    virtual string readMessage(HANDLE hSerial) {
        return "Hello, World!";
    }
};

class MockSerialPort : public SerialPort {
public:
    HANDLE openSerialPort(const wstring& portName, DWORD baudRate) override {
        if (portName == L"COM1") {
            return INVALID_HANDLE_VALUE;
        }
        return (HANDLE)1;
    }
    void sendMessage(HANDLE hSerial, const string& message) override {
        SetLastError(ERROR_IO_DEVICE);
    }

    string readMessage(HANDLE hSerial) override {
        return "Hello, World!";
    }
};

TEST(SerialPortTest, OpenSerialPortFails) {
    MockSerialPort mockSerial;
    HANDLE handle = mockSerial.openSerialPort(L"COM1", 9600);
    EXPECT_EQ(handle, INVALID_HANDLE_VALUE);
}

TEST(SerialPortTest, SendMessageFails) {
    MockSerialPort mockSerial;
    HANDLE mockHandle = (HANDLE)1;
    string message = "Hello, World!";

    mockSerial.sendMessage(mockHandle, message);
    EXPECT_EQ(GetLastError(), ERROR_IO_DEVICE);
}

TEST(SerialPortTest, ReadMessageReturnsCorrectData) {
    MockSerialPort mockSerial;
    HANDLE mockHandle = (HANDLE)1;
    string expectedMessage = "Hello, World!";
    string actualMessage = mockSerial.readMessage(mockHandle);
    EXPECT_EQ(actualMessage, expectedMessage);
}
